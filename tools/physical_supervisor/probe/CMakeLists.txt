cmake_minimum_required(VERSION 3.20)
project(sdvk009_instance_probe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This probe requires the reviewed native Windows x64 MSVC ABI")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(target sdvk009_instance_probe sdvk009_probe_cpu_tests)
  if(target STREQUAL "sdvk009_instance_probe")
    add_executable(${target} ../instance_probe.cpp)
  else()
    add_executable(${target} ../test_instance_probe.cpp)
  endif()
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../libraries/ZVulkan/include")
endforeach()
